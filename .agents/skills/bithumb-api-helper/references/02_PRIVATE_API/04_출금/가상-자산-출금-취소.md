# 가상 자산 출금 취소

가상 자산 출금을 취소합니다.


> #### 안내 사항
> 
> 처리 중(`PROCESSING`) 상태의 출금에 대해  취소가 가능하며, 이미 출금이 완료(`DONE`)되거나 취소(`CANCELLED`)된 상태일 경우, 출금 취소가 불가합니다.


---

## 가상 자산 출금 취소

가상 자산 출금을 취소합니다.

| 항목 | 내용 |
|------|------|
| **Method** | `DELETE` |
| **URL** | `https://api.bithumb.com/v1/withdraws/coin` |
| **인증** | ✅ JWT Bearer 토큰 필요 |
| **Content-Type** | `application/json` |

### 요청 파라미터

#### 쿼리 파라미터

| 파라미터 | 타입 | 필수 | 설명 |
|--------|------|------|------|
| `withdrawal_id` | string | ✅ | 출금의 고유 ID |

### 응답

#### `200` ✅ 성공

**응답 예시**

```json
"{\n  \"type\": \"withdraw\",\n  \"withdrawal_id\": \"200377211\",\n  \"currency\": \"BTC\",\n  \"net_type\": \"BTC\",\n  \"state\": \"PROCESSING\",\n  \"created_at\": \"2024-07-14T14:54:24+09:00\",\n  \"amount\": \"0.00010000\",\n  \"fee\": \"0\",\n  \"krw_amount\": \"8400\"\n}"
```

**응답 필드**

| 필드 | 타입 | 필수 | 설명 |
|------|------|------|------|
| `type` | string | ✅ | 입출금 종류 |
| `withdrawal_id` | string | ✅ | 출금의 고유 ID |
| `currency` | string | ✅ | 화폐 심볼 (예시: BTC) |
| `net_type` | string | ✅ | 출금 네트워크 |
| `state` | string | ✅ | 출금 상태 |
| `created_at` | string | ✅ | 출금 생성 시각 |
| `amount` | string | ✅ | 출금 금액/수량 |
| `fee` | string | ✅ | 출금 수수료 |
| `krw_amount` | string | ✅ | 원화 환산 가격 |

#### `400` ❌ 오류

**응답 예시**

```json
"{\n  \"error\": {\n    \"name\": \"invalid_parameter\",\n    \"message\": \"\\\"withdrawal_id\\\" is missing\"\n  }\n}"
```

**응답 필드**

| 필드 | 타입 | 필수 | 설명 |
|------|------|------|------|
| `error` | object | ✅ |  |

// Required packages:

const jwt = require('jsonwebtoken');
const { v4: uuidv4 } = require('uuid');
const crypto = require('crypto');
const axios = require('axios');

// TODO: Replace with your API key.
const accessKey = 'YOUR_ACCESS_KEY';
// TODO: Replace with your secret key.
const secretKey = 'YOUR_SECRET_KEY';
// TODO: Replace with the withdrawal ID to cancel.
const withdrawalId = 'YOUR_WITHDRAWAL_ID';

const apiUrl = 'https://api.bithumb.com';
const algorithm = 'SHA512';

function createPayload(queryString) {
  const queryHash = crypto.createHash(algorithm).update(queryString, 'utf-8').digest('hex');

  return {
    access_key: accessKey,
    nonce: uuidv4(),
    timestamp: Date.now(),
    query_hash: queryHash,
    query_hash_alg: algorithm,
  };
}

async function main() {
  const params = { withdrawal_id: withdrawalId };
  const queryString = new URLSearchParams(params).toString();
  const payload = createPayload(queryString);
  const jwtToken = jwt.sign(payload, secretKey);

  const response = await axios.delete(`${apiUrl}/v1/withdraws/coin`, {
    headers: {
      Authorization: `Bearer ${jwtToken}`,
    },
    params,
  });

  console.log('status:', response.status);
  console.log('data:', response.data);
}

main()
  .catch((error) => {
    console.log('status:', error.response?.status ?? 'NO_RESPONSE');
    console.log('data:', error.response?.data ?? error.message);
  });

```python
import base64
import hashlib
import hmac
import json
import time
import urllib.error
import urllib.parse
import urllib.request
import uuid

# TODO: Replace with your API key.
ACCESS_KEY = "YOUR_ACCESS_KEY"
# TODO: Replace with your secret key.
SECRET_KEY = "YOUR_SECRET_KEY"
# TODO: Replace with the withdrawal ID to cancel.
WITHDRAWAL_ID = "YOUR_WITHDRAWAL_ID"

API_URL = "https://api.bithumb.com"
ALGORITHM = "SHA512"

def to_base64_url(value: bytes) -> str:
    return base64.urlsafe_b64encode(value).rstrip(b"=").decode("utf-8")

def create_jwt(payload: dict[str, str | int], secret_key: str) -> str:
    header = {"alg": "HS256", "typ": "JWT"}
    encoded_header = to_base64_url(json.dumps(header, separators=(",", ":")).encode("utf-8"))
    encoded_payload = to_base64_url(json.dumps(payload, separators=(",", ":")).encode("utf-8"))
    signature = hmac.new(
        secret_key.encode("utf-8"),
        f"{encoded_header}.{encoded_payload}".encode("utf-8"),
        hashlib.sha256,
    ).digest()
    encoded_signature = to_base64_url(signature)
    return f"{encoded_header}.{encoded_payload}.{encoded_signature}"

def create_payload(query_string: str) -> dict[str, str | int]:
    query_hash = hashlib.sha512(query_string.encode("utf-8")).hexdigest()
    return {
        "access_key": ACCESS_KEY,
        "nonce": str(uuid.uuid4()),
        "timestamp": int(time.time() * 1000),
        "query_hash": query_hash,
        "query_hash_alg": ALGORITHM,
    }

def main() -> None:
    params = {"withdrawal_id": WITHDRAWAL_ID}
    query_string = urllib.parse.urlencode(params)
    payload = create_payload(query_string)
    jwt_token = create_jwt(payload, SECRET_KEY)

    request = urllib.request.Request(
        url=f"{API_URL}/v1/withdraws/coin?{query_string}",
        method="DELETE",
        headers={
            "Authorization": f"Bearer {jwt_token}",
            "accept": "application/json",
        },
    )

    try:
        with urllib.request.urlopen(request) as response:
            response_body = response.read().decode("utf-8")
            print("status:", response.status)
            try:
                print("data:", json.loads(response_body))
            except json.JSONDecodeError:
                print("data:", response_body)
    except urllib.error.HTTPError as error:
        response_body = error.read().decode("utf-8", errors="replace")
        print("status:", error.code)
        try:
            print("data:", json.loads(response_body))
        except json.JSONDecodeError:
            print("data:", response_body)
    except urllib.error.URLError as error:
        print("status:", "NO_RESPONSE")
        print("data:", str(error.reason))


if __name__ == "__main__":
    main()
```

import java.io.IOException;
import java.net.URI;
import java.net.http.HttpClient;
import java.net.http.HttpRequest;
import java.net.http.HttpResponse;
import java.nio.charset.StandardCharsets;
import java.security.MessageDigest;
import java.time.Instant;
import java.util.Base64;
import java.util.UUID;

import javax.crypto.Mac;
import javax.crypto.spec.SecretKeySpec;

public final class DeleteWithdrawCoin {
    // TODO: Replace with your API key.
    private static final String ACCESS_KEY = "YOUR_ACCESS_KEY";
    // TODO: Replace with your secret key.
    private static final String SECRET_KEY = "YOUR_SECRET_KEY";
    // TODO: Replace with the withdrawal ID to cancel.
    private static final String WITHDRAWAL_ID = "YOUR_WITHDRAWAL_ID";

    private static final String API_URL = "https://api.bithumb.com";
    private static final String QUERY_HASH_ALGORITHM = "SHA512";

    private DeleteWithdrawCoin() {
    }

    public static void main(String[] args) throws Exception {
        String queryString = "withdrawal_id=" + WITHDRAWAL_ID;
        String queryHash = sha512Hex(queryString);
        String payload = "{"
            + "\"access_key\":\"" + ACCESS_KEY + "\","
            + "\"nonce\":\"" + UUID.randomUUID() + "\","
            + "\"timestamp\":" + Instant.now().toEpochMilli() + ","
            + "\"query_hash\":\"" + queryHash + "\","
            + "\"query_hash_alg\":\"" + QUERY_HASH_ALGORITHM + "\""
            + "}";
        String jwtToken = createJwt(payload, SECRET_KEY);

        HttpRequest request = HttpRequest.newBuilder()
            .uri(URI.create(API_URL + "/v1/withdraws/coin?" + queryString))
            .header("Authorization", "Bearer " + jwtToken)
            .header("accept", "application/json")
            .DELETE()
            .build();

        try (HttpClient client = HttpClient.newHttpClient()) {
            HttpResponse<String> response = client.send(request, HttpResponse.BodyHandlers.ofString());
            System.out.println("status: " + response.statusCode());
            System.out.println("data: " + response.body());
        } catch (IOException | InterruptedException error) {
            System.out.println("status: NO_RESPONSE");
            System.out.println("data: " + error.getMessage());
            if (error instanceof InterruptedException) {
                Thread.currentThread().interrupt();
            }
        }
    }

    private static String createJwt(String payload, String secretKey) throws Exception {
        String header = "{\"alg\":\"HS256\",\"typ\":\"JWT\"}";
        String encodedHeader = toBase64Url(header.getBytes(StandardCharsets.UTF_8));
        String encodedPayload = toBase64Url(payload.getBytes(StandardCharsets.UTF_8));
        String signatureInput = encodedHeader + "." + encodedPayload;

        Mac mac = Mac.getInstance("HmacSHA256");
        mac.init(new SecretKeySpec(secretKey.getBytes(StandardCharsets.UTF_8), "HmacSHA256"));
        byte[] signature = mac.doFinal(signatureInput.getBytes(StandardCharsets.UTF_8));

        return signatureInput + "." + toBase64Url(signature);
    }

    private static String sha512Hex(String value) throws Exception {
        MessageDigest digest = MessageDigest.getInstance("SHA-512");
        byte[] hash = digest.digest(value.getBytes(StandardCharsets.UTF_8));
        StringBuilder builder = new StringBuilder();
        for (byte current : hash) {
            builder.append(String.format("%02x", current));
        }
        return builder.toString();
    }

    private static String toBase64Url(byte[] value) {
        return Base64.getUrlEncoder().withoutPadding().encodeToString(value);
    }
}
